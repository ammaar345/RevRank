# Firestore Security Rules — firestore.rules

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isSignedIn() { return request.auth != null; }
    function isOwner(uid) { return request.auth.uid == uid; }
    function isPro(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data.isPro == true;
    }
    
    // Users — read own, write own only
    match /users/{uid} {
      allow read: if isSignedIn() && (isOwner(uid) || resource.data.keys().hasAll(['username', 'rank']));
      allow write: if isSignedIn() && isOwner(uid);
      
      // Referrals subcollection
      match /referrals/{refId} {
        allow read: if isSignedIn() && isOwner(uid);
        allow write: if false; // Cloud Functions only
      }
    }
    
    // Username uniqueness claims
    match /usernames/{username} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow delete: if isSignedIn() && resource.data.uid == request.auth.uid;
      allow update: if false;
    }
    
    // Trips — owner only for full data, public for challenge context only
    match /trips/{tripId} {
      allow read: if isSignedIn() && (
        resource.data.userId == request.auth.uid ||
        resource.data.isPublic == true  // only score, grade, distance exposed
      );
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow delete: if isSignedIn() && resource.data.userId == request.auth.uid;
    }
    
    // Weekly scores — read all (for leaderboard), write by Cloud Functions only
    match /weeklyScores/{uid} {
      allow read: if isSignedIn();
      allow write: if false; // Cloud Functions only
    }
    
    // Challenges
    match /routeChallenges/{challengeId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.resource.data.creatorUid == request.auth.uid;
      allow update: if isSignedIn(); // anyone can accept (add to acceptances array)
      allow delete: if isSignedIn() && resource.data.creatorUid == request.auth.uid;
    }
    
    // Weekly challenges (generated by Cloud Functions)
    match /challenges/{uid} {
      allow read: if isSignedIn() && isOwner(uid);
      allow write: if false; // Cloud Functions only
    }
    
    // Badges
    match /badges/{badgeId} {
      allow read: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if false;
    }
  }
}
```
